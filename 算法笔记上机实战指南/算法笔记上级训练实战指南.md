### B1046划拳

failA,failB记录各自输的次数

```c++
#include <iostream>
using namespace std;
int failA, failB;
int n;

int main()
{
    cin >> n;
    while(n --)
    {
        int a1, a2, b1, b2;
        cin >> a1 >> a2 >> b1 >> b2;
        if(a1 + b1 == a2 && a1 + b1 != b2) failB ++;
        if(a1 + b1 == b2 && a1 + b1 != a2) failA ++;
    }
    cout << failA << " " << failB << endl;

    return 0;
}
```

### B1008数组元素循环右移问题

:warning:题目没有给定M的最大值，不能认为M<N。读入后需要令M=M%N。

直接输出N-M到N-1号元素，再输出0—N - M - 1号元素。

```c++
#include <iostream>
using namespace std;
const int N = 110;
int a[N];
int n, m;

int main()
{
    scanf("%d %d", &n, &m);
    m %= n;
    for(int i = 0; i < n; i ++)
        scanf("%d", &a[i]);
    for(int i = n - m; i < n; i ++)
        cout << a[i] << " ";
    for(int i = 0; i < n - m; i ++)
        cout << a[i] << " \n"[ i == n - m - 1];
    return 0;
}
```
