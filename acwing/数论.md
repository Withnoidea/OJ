## 质数判定-试除法

O(sqrt(n))

```c++
bool is_prime(int n)
{
	if(n < 2) return false;
	for(int i = 2; i <= n / i; i ++)
		if(n % i == 0)
			return false;
	return true;
}
```

int get_prime(int n)
{
    for(int i = 2; i <= n; i ++)
    {
        if(!st[i]) prime[cnt ++] = i;
        for(int j = 0; prime[j] <= n / i; j ++)
        {
            st[prime[j] * i] = true;
            if(i % prime[j] == 0) break;
        }
    }
    return cnt;
}

## 欧拉定理

欧拉函数的定义

 $1 \sim N$ 中与 $N$ 互质的数的个数被称为欧拉函数，记为 $ϕ(N)$。

若在算数基本定理中，$N = p_1^{a_1}p_2^{a_2}…p_m^{a_m}$，则：$$ϕ(N) = N \times \frac{p_1-1}{p_1} \times \frac{p_2-1}{p_2} \times … \times \frac{p_m-1}{p_m}$$

![image-20250608141140601](https://cdn.jsdelivr.net/gh/Withnoidea/picgo@main/images/image-20250608141140601.png)

```
#include <bits/stdc++.h>
using namespace std;
int n;

int main()
{
    cin >> n;
    while(n --)
    {
        int a;
        cin >> a;
        int res = a;
        for(int i = 2; i <= a / i; i ++)
        {
            if(a % i == 0)
            {
                res = res / i * (i - 1);
                while(a % i == 0) a /= i;
            }
        }
        if(a > 1) res = res / a * (a - 1);
        cout << res << endl;
    }
    
    return 0;
}
```

## 线性筛求欧拉函数

```
#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
const int N = 1e6 + 10;
int primes[N];
bool st[N];
int phi[N];
int cnt;
int n;

LL get_eula(int n)
{
    phi[1] = 1;
    for(int i = 2; i <= n; i ++)
    {
        if(!st[i])
        {
            primes[cnt ++] = i;
            phi[i] = i - 1;
        }
        for(int j = 0; primes[j] <= n / i; j ++)
        {
            st[primes[j] * i] = true;
            if(i % primes[j] == 0)
            {
                phi[primes[j] * i] = phi[i] * primes[j];
                break;
            }
            phi[primes[j] * i] = phi[i] * (primes[j] - 1);
        }
    }
    
    LL res = 0;   
    for(int i = 1; i <= n; i ++) res += phi[i];
    
    return res;
}

int main()
{
    cin >> n;
    cout << get_eula(n) << endl;
    
    return 0;
}
```

## 快速幂求逆元

乘法逆元的定义

若整数 $b，m$ 互质，并且对于任意的整数 $a$，如果满足 $b|a$，则存在一个整数 $x$，使得 $ \frac{a}{b} ≡a \times x \pmod m$ ，则称 $x$ 为 $b$ 的模 $m$ 乘法逆元，记为 $b^{-1} \pmod m$。

 $b$ 存在乘法逆元的充要条件是 $b$ 与模数 $m$ 互质。当模数 $m$ 为质数时，$b^{m-2}$ 即为 $b$ 的乘法逆元。
